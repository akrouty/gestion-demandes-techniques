% Les sources des chapitres conservent leurs sauts historiques. Lors de leur
% inclusion, le second \clearpage et le second \markboth sont neutralisés :
% la première page contient ainsi le titre et l'introduction, tandis que les
% autres sauts de page voulus à l'intérieur du chapitre restent actifs.
\newcount\ReportClearpageCall
\newcount\ReportMarkCall
\newcommand{\ReportIncludeChapter}[1]{%
  \ReportClearpageCall=0
  \ReportMarkCall=0
  \let\ReportSavedClearpage\clearpage
  \let\ReportSavedMarkboth\markboth
  \def\clearpage{%
    \advance\ReportClearpageCall by 1
    \ifnum\ReportClearpageCall=2\relax\pagestyle{reportbody}%
    \else\ReportSavedClearpage\fi}%
  \def\markboth##1##2{%
    \advance\ReportMarkCall by 1
    \ifnum\ReportMarkCall=2\relax
    \else\ReportSavedMarkboth{##1}{##2}\fi}%
  \input{#1}%
  \let\clearpage\ReportSavedClearpage
  \let\markboth\ReportSavedMarkboth
}
